// Scene Setup

const scene = new THREE.Scene();

scene.fog = new THREE.FogExp2(0x030508, 0.035);


const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);

camera.position.set(0, 14, 20);


const renderer = new THREE.WebGLRenderer({ antialias: true });

renderer.setSize(window.innerWidth, window.innerHeight);

renderer.setPixelRatio(window.devicePixelRatio);

document.body.appendChild(renderer.domElement);


const controls = new THREE.OrbitControls(camera, renderer.domElement);

controls.enableDamping = true;

controls.dampingFactor = 0.05;


// Lighting

const ambientLight = new THREE.AmbientLight(0xffffff, 0.8);

scene.add(ambientLight);


const pointLight = new THREE.PointLight(0x33ff88, 2.5, 50);

pointLight.position.set(0, 10, 0);

scene.add(pointLight);


// 1. Central Sgr A* Core Node (High-Capacity Lattice Knot)

const coreGeo = new THREE.SphereGeometry(1.0, 32, 32);

const coreMat = new THREE.MeshStandardMaterial({ color: 0xffa657, emissive: 0xffa657, emissiveIntensity: 0.8 });

const sgrA = new THREE.Mesh(coreGeo, coreMat);

sgrA.position.set(0, 0, 0);

scene.add(sgrA);


// 2. Star S2 Orbiting the Core

const s2Geo = new THREE.SphereGeometry(0.4, 32, 32);

const s2Mat = new THREE.MeshStandardMaterial({ color: 0x79c0ff, emissive: 0x79c0ff, emissiveIntensity: 0.9 });

const s2Star = new THREE.Mesh(s2Geo, s2Mat);

scene.add(s2Star);


// 3. Orbital Path Ring

const orbitRadiusX = 7.0;

const orbitRadiusZ = 4.5;

const curvePoints = [];

for (let i = 0; i <= 100; i++) {

    let theta = (i / 100) * Math.PI * 2;

    curvePoints.push(new THREE.Vector3(Math.cos(theta) * orbitRadiusX, 0, Math.sin(theta) * orbitRadiusZ));

}

const orbitGeo = new THREE.BufferGeometry().setFromPoints(curvePoints);

const orbitMat = new THREE.LineBasicMaterial({ color: 0x30363d });

const orbitLine = new THREE.Line(orbitGeo, orbitMat);

scene.add(orbitLine);


// 4. Deformable Pseudomanifold Grid

const gridDim = 24;

const planeGeo = new THREE.PlaneGeometry(gridDim, gridDim, 28, 28);

planeGeo.rotateX(-Math.PI / 2);


const planeMat = new THREE.MeshStandardMaterial({

    color: 0x05140a,

    emissive: 0x22ee77,

    emissiveIntensity: 0.5,

    wireframe: true,

    side: THREE.DoubleSide

});

const gridMesh = new THREE.Mesh(planeGeo, planeMat);

scene.add(gridMesh);


const posAttribute = planeGeo.attributes.position;

const initialY = [];

for (let i = 0; i < posAttribute.count; i++) {

    initialY.push(posAttribute.getY(i));

}


// Animation Loop

let clock = 0;

function animate() {

    requestAnimationFrame(animate);

    clock += 0.010;


    // Elliptical orbit calculation for S2

    let angle = clock * 0.6;

    let s2X = Math.cos(angle) * orbitRadiusX;

    let s2Z = Math.sin(angle) * orbitRadiusZ;

    s2Star.position.set(s2X, 0, s2Z);


    // Dynamic grid lattice deformation around Sgr A* and S2

    for (let i = 0; i < posAttribute.count; i++) {

        let vx = posAttribute.getX(i);

        let vz = posAttribute.getZ(i);

        

        let distCore = Math.sqrt(vx * vx + vz * vz);

        let distS2 = Math.sqrt(Math.pow(vx - s2X, 2) + Math.pow(vz - s2Z, 2));

        

        let coreDeflection = Math.max(0, (3.5 - distCore)) * 0.8;

        let s2Deflection = Math.max(0, (1.5 - distS2)) * 0.3;

        

        posAttribute.setY(i, initialY[i] - coreDeflection - s2Deflection);

    }

    posAttribute.needsUpdate = true;


    controls.update();

    renderer.render(scene, camera);

}


animate();


window.addEventListener('resize', () => {

    camera.aspect = window.innerWidth / window.innerHeight;

    camera.updateProjectionMatrix();

    renderer.setSize(window.innerWidth, window.innerHeight);

});
